# htsim option parameter
--expt:
    val=1 # flow convergence
    val=2 # single link simulation
    val=3 # conga
    val=4 # pfabric
    val=5 # incast
    val=6 # fat tree

--flowdist:
    val=pareto
    val=enterprise
    val=datamining
    val=<null> # uniform

--queue:
    val=fq # fair queue
    val=cq # calendar queue
    val=afq # approximate fair queue
    val=pq # priority queue
    val=sfq # stocastic fair queue
    val=<null> # fifo queue

--endhost:
    val=pp # packet pair
    val=timely
    val=dctcp
    val=dtcp
    val=ddctcp

--logfile=: # log file
--utilization: # faction number (0, 1)

# autoscale weight-transfer parameters (--expt=4)
--scale_mode:
    val=fixed # default: fixed timer boot delay (--bootdelay)
    val=transfer # model weight transfer over the fabric using CSIG

--transfer_mode: # used when --scale_mode=transfer
    val=k_peers # default: k active donors chosen by capacity WRR, chunks distributed across cores via CSIG WRR
    val=single # 1 highest-capacity active donor sends all chunks to booting nodes via CSIG WRR
    val=relay # 1 initial active donor -> fanout F booting nodes -> pipelined relay tree via CSIG WRR

--model_size: # simulated full LLM parameter set size in MB (default: 50.0)
--model_fraction: # fraction of model_size transferred per booting node (default: 0.10)
--transfer_k: # number of active donor nodes in k_peers mode (default: 4)
--transfer_fanout: # number of child booting nodes per parent in relay mode (default: 2)
--chunk_size: # size of each parameter transfer chunk in bytes (default: 100000)

# log format
## type1: flow start
<flow name> <start time> <flow ID> <flow size> <src_node> <des_nod>
## type2: flow finish
Flow <flow name> <flow ID> size <flow size> start <start time> end <end time> fct <flow completion time> sent <round to MTU> tput <throughput> rtt <RTT time> cwnd <congestion window size> alpha <alpha value>
## type3: queue stat
<queue name> <simulation time> stats <flow id>-><packet number> ...
